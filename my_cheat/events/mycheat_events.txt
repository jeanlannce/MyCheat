# My Cheat 事件：科研突破点 + 战略资源
# 注意：namespace 必须声明（否则事件 id 无法解析）

add_namespace = mycheat

##################################################################
### 科研突破点（入口）                                          ###
### 决议触发本事件，先选学科，再选档位                          ###
##################################################################
country_event = {
	id = breakthrough
	title = mycheat.breakthrough.t
	desc = mycheat.breakthrough.d
	picture = GFX_report_event_generic_research_lab

	is_triggered_only = yes

	option = {
		name = mycheat.breakthrough.opt_land
		country_event = { id = breakthrough_land }
	}
	option = {
		name = mycheat.breakthrough.opt_naval
		country_event = { id = breakthrough_naval }
	}
	option = {
		name = mycheat.breakthrough.opt_air
		country_event = { id = breakthrough_air }
	}
	option = {
		name = mycheat.breakthrough.opt_nuclear
		country_event = { id = breakthrough_nuclear }
	}
	option = {
		name = mycheat.breakthrough.opt_all
		country_event = { id = breakthrough_all }
	}
}

country_event = {
	id = mining
	title = mycheat.mining.t
	desc = mycheat.mining.d
	picture = GFX_report_event_generic_factory

	is_triggered_only = yes

	option = {
		name = mycheat.mining.small
		capital_scope = {
			add_resource = { type = oil  amount = 50 }
			add_resource = { type = steel  amount = 50 }
			add_resource = { type = aluminium  amount = 50 }
			add_resource = { type = rubber  amount = 50 }
			add_resource = { type = tungsten  amount = 50 }

		}
	}
	option = {
		name = mycheat.mining.medium
		capital_scope = {
			add_resource = { type = oil  amount = 100 }
			add_resource = { type = steel  amount = 100 }
			add_resource = { type = aluminium  amount = 100 }
			add_resource = { type = rubber  amount = 100 }
			add_resource = { type = tungsten  amount = 100 }
			add_resource = { type = chromium  amount = 100 }
			add_resource = { type = coal  amount = 100 }
		}
	}
	option = {
		name = mycheat.mining.huge
		capital_scope = {
			add_resource = { type = oil  amount = 50000 }
			add_resource = { type = steel  amount = 50000 }
			add_resource = { type = aluminium  amount = 50000 }
			add_resource = { type = rubber  amount = 50000 }
			add_resource = { type = tungsten  amount = 50000 }
			add_resource = { type = chromium  amount = 50000 }
			add_resource = { type = coal  amount = 50000 }
			### 若你的其他 mod 新增了资源，在此处复制一行：
			### add_resource = { type = <资源名>  amount = N }
		}
	}
}
